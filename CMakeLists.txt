cmake_minimum_required(VERSION 3.16)
project(ember CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EMBER_LINKED_STACK "Use linked Stack implementation" OFF)
if(EMBER_LINKED_STACK)
    set(STACK_SOURCE src/stack_linked.cpp)
else()
    set(STACK_SOURCE src/stack.cpp)
endif()
set(EMBER_CORE src/memory.cpp src/dump.cpp src/alu.cpp src/cpu.cpp src/display.cpp src/sprite.cpp src/asm/token.cpp src/asm/lexer.cpp src/asm/assembler.cpp src/asm/frontend.cpp ${STACK_SOURCE})
add_executable(ember src/main.cpp ${EMBER_CORE})
include(CTest)
if(BUILD_TESTING)
    add_executable(alu-check tests/alu_checks.cpp ${EMBER_CORE})
    target_include_directories(alu-check PRIVATE src)
    add_test(NAME alu-and-cpu COMMAND alu-check)
    add_executable(stack-check tests/stack_checks.cpp ${EMBER_CORE})
    target_include_directories(stack-check PRIVATE src)
    add_test(NAME stack-and-calls COMMAND stack-check)
    add_executable(assembler-check tests/assembler_checks.cpp ${EMBER_CORE})
    target_include_directories(assembler-check PRIVATE src)
    add_test(NAME assembler-and-lexer COMMAND assembler-check)
    add_executable(layout tests/layout.cpp)
    target_include_directories(layout PRIVATE src)
    add_test(NAME structure-layout COMMAND layout)
endif()
foreach(target ember alu-check layout stack-check assembler-check)
    if(TARGET ${target})
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        if(CMAKE_BUILD_TYPE STREQUAL "Debug" AND NOT MSVC)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()

if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        foreach(check check_cli check_machine check_control check_display check_records check_stack check_assembly)
            add_test(NAME ${check} COMMAND ${Python3_EXECUTABLE} tests/${check}.py $<TARGET_FILE:ember>)
            set_tests_properties(${check} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
        endforeach()
    endif()
endif()
